VERSION = 1,

/*
 * Server controlled balance for QoL Compendium.
 *
 * Everything here is balance, so it belongs to the server rather than the individual
 * player. Mod options are per client and would let one player quietly run different
 * numbers to everyone else on the same server. Cosmetic settings go there instead.
 *
 * One page for the whole compendium, matching the single mod options category, so a
 * host sees one QoL Compendium heading rather than one per absorbed mod.
 *
 * Read from lua as SandboxVars.QoLC.<name>. Labels live in
 * 42/media/lua/shared/Translate/EN/Sandbox.json, keyed Sandbox_<translation>.
 */

option QoLC.AdrenalineEnabled
{
    type = boolean, default = true,
    page = QoLC,
    translation = QoLC_AdrenalineEnabled,
}

option QoLC.AdrenalineBoostSpeed
{
    type = integer, min = 1, max = 500, default = 75,
    page = QoLC,
    translation = QoLC_AdrenalineBoostSpeed,
}

option QoLC.AdrenalineCrashSpeed
{
    type = integer, min = 1, max = 500, default = 70,
    page = QoLC,
    translation = QoLC_AdrenalineCrashSpeed,
}

option QoLC.AdrenalineCrashPenalty
{
    type = double, min = 1.001, max = 2.0, default = 1.015,
    page = QoLC,
    translation = QoLC_AdrenalineCrashPenalty,
}

/*
 * Pump units to fill one propane tank from empty. A pump holds 22000 and a jerry can
 * takes 10, so the default of 20 makes a tank cost two cans of fuel and leaves a full
 * pump good for around a thousand of them.
 */
option QoLC.PropanePumpCost
{
    type = integer, min = 1, max = 2000, default = 20,
    page = QoLC,
    translation = QoLC_PropanePumpCost,
}

/*
 * Tailoring experience for cutting a garment into strips, which vanilla pays nothing at
 * all for. Off returns ripping to vanilla, both the experience and the speed it gains
 * with the skill. The rate scales what is awarded, so a host can grind it slower or
 * faster without turning it off.
 */
option QoLC.TailoringCutXpEnabled
{
    type = boolean, default = true,
    page = QoLC,
    translation = QoLC_TailoringCutXpEnabled,
}

option QoLC.TailoringCutXp
{
    type = integer, min = 0, max = 500, default = 100,
    page = QoLC,
    translation = QoLC_TailoringCutXp,
}

/*
 * A house whose door already stands open, or whose window is smashed, does not sound its
 * alarm. Three settings rather than a switch, because the last step is much broader than
 * it reads: vanilla leaves most exterior doors unlocked, so counting an unlocked door
 * disarms far more houses than an open one does. That is the original mod's own
 * behaviour and the default here, but a host who wants alarms to still mean something
 * can step it back one.
 */
option QoLC.AlarmHouseRule
{
    type = enum, numValues = 3, default = 3,
    page = QoLC,
    translation = QoLC_AlarmHouseRule,
    valueTranslation = QoLC_AlarmHouseRule,
}

option QoLC.AlarmVehiclesEnabled
{
    type = boolean, default = true,
    page = QoLC,
    translation = QoLC_AlarmVehiclesEnabled,
}

/*
 * Reading a skill book eases boredom, unhappiness and stress. Each page takes a share of
 * what is left rather than a flat amount, so it settles a character without ever
 * finishing the job, and the rate scales the lot for a host who wants it gentler. This
 * is a roleplay preference more than a fix, which is why it is one switch away from off.
 */
option QoLC.ReadingCalmEnabled
{
    type = boolean, default = true,
    page = QoLC,
    translation = QoLC_ReadingCalmEnabled,
}

option QoLC.ReadingCalmRate
{
    type = integer, min = 0, max = 500, default = 100,
    page = QoLC,
    translation = QoLC_ReadingCalmRate,
}

/*
 * Sleeping wears off boredom and unhappiness. The base game freezes both while a
 * character is asleep, so a night changes neither. The rate scales what a night is
 * worth: at the default, eight hours clears boredom outright.
 */
option QoLC.SleepRecoveryEnabled
{
    type = boolean, default = true,
    page = QoLC,
    translation = QoLC_SleepRecoveryEnabled,
}

option QoLC.SleepRecoveryRate
{
    type = integer, min = 0, max = 500, default = 100,
    page = QoLC,
    translation = QoLC_SleepRecoveryRate,
}

/*
 * A magazine that teaches the Nutritionist trait, so food tooltips start showing
 * calories and the rest. The base game has no way to learn a trait after character
 * creation at all, so this is the only route besides spending two points at the start or
 * taking Fitness Instructor. Off stops it spawning; a copy already found still works.
 */
option QoLC.NutritionistMagEnabled
{
    type = boolean, default = true,
    page = QoLC,
    translation = QoLC_NutritionistMagEnabled,
}

/*
 * Lockpicking, which the base game has none of: no picks, no recipes, and nothing on a
 * door that opens a lock without its key. Off stops the books spawning and takes the
 * options off the menu; picks already made stay in the world as junk.
 */
option QoLC.LockpickingEnabled
{
    type = boolean, default = true,
    page = QoLC,
    translation = QoLC_LockpickingEnabled,
}

/*
 * Switches for the three features that change balance rather than presentation. They sit
 * here rather than in mod options for the reason the rest of this page exists: a per
 * client setting would let one player on a server refill propane, weld twice as long, or
 * find slings in loot while another cannot.
 *
 * Turning the sling off stops it spawning in loot and on dead soldiers. Its recipe and
 * its clothing items are script definitions rather than lua, so they always exist and
 * anything already owned keeps working.
 */
option QoLC.SlingEnabled
{
    type = boolean, default = true,
    page = QoLC,
    translation = QoLC_SlingEnabled,
}

option QoLC.PropanePumpEnabled
{
    type = boolean, default = true,
    page = QoLC,
    translation = QoLC_PropanePumpEnabled,
}

option QoLC.BlowtorchEnabled
{
    type = boolean, default = true,
    page = QoLC,
    translation = QoLC_BlowtorchEnabled,
}

/*
 * Cutting up the garments build 42 never tagged as cuttable: underwear, socks, bandanas,
 * scarves, gloves, hide trousers and shoes. Off returns all ninety two to being uncuttable
 * and leaves everything vanilla already allowed exactly as it was.
 *
 * Balance rather than presentation, because it changes how much cloth and leather a world
 * holds. Shoes are the reason: they are everywhere, and most of them are leather.
 */
option QoLC.CutClothingEnabled
{
    type = boolean, default = true,
    page = QoLC,
    translation = QoLC_CutClothingEnabled,
}
