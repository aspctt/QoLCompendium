VERSION = 1,

/*
 * Server controlled balance for QoL Compendium.
 *
 * Everything here is balance, so it belongs to the server rather than the individual
 * player. Mod options are per client and would let one player quietly run different
 * numbers to everyone else on the same server. Cosmetic settings go there instead.
 *
 * One page for the whole compendium, matching the single mod options category, so a
 * host sees one QoL Compendium heading rather than one per absorbed mod.
 *
 * Read from lua as SandboxVars.QoLC.<name>. Labels live in
 * 42/media/lua/shared/Translate/EN/Sandbox.json, keyed Sandbox_<translation>.
 */

option QoLC.AdrenalineEnabled
{
    type = boolean, default = true,
    page = QoLC,
    translation = QoLC_AdrenalineEnabled,
}

option QoLC.AdrenalineBoostSpeed
{
    type = integer, min = 1, max = 500, default = 75,
    page = QoLC,
    translation = QoLC_AdrenalineBoostSpeed,
}

option QoLC.AdrenalineCrashSpeed
{
    type = integer, min = 1, max = 500, default = 70,
    page = QoLC,
    translation = QoLC_AdrenalineCrashSpeed,
}

option QoLC.AdrenalineCrashPenalty
{
    type = double, min = 1.001, max = 2.0, default = 1.015,
    page = QoLC,
    translation = QoLC_AdrenalineCrashPenalty,
}

/*
 * Pump units to fill one propane tank from empty. A pump holds 22000 and a jerry can
 * takes 10, so the default of 20 makes a tank cost two cans of fuel and leaves a full
 * pump good for around a thousand of them.
 */
option QoLC.PropanePumpCost
{
    type = integer, min = 1, max = 2000, default = 20,
    page = QoLC,
    translation = QoLC_PropanePumpCost,
}

/*
 * Tailoring experience for cutting a garment into strips, which vanilla pays nothing at
 * all for. Off returns ripping to vanilla, both the experience and the speed it gains
 * with the skill. The rate scales what is awarded, so a host can grind it slower or
 * faster without turning it off.
 */
option QoLC.TailoringCutXpEnabled
{
    type = boolean, default = true,
    page = QoLC,
    translation = QoLC_TailoringCutXpEnabled,
}

option QoLC.TailoringCutXp
{
    type = integer, min = 0, max = 500, default = 100,
    page = QoLC,
    translation = QoLC_TailoringCutXp,
}
